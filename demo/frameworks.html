🚀 Framework Integration Examples

Vanilla JavaScript

// Create element programmatically
const qr = document.createElement('qr-code');
qr.setAttribute('data-text', 'Hello World');
qr.setAttribute('data-width', '256');
qr.setAttribute('data-height', '256');

// Or use properties
qr.text = 'Hello World';
qr.width = 256;
qr.height = 256;

// Listen for events
qr.addEventListener('qrcode-ready', (event) => {
  console.log('QR Code generated:', event.detail);
});

document.body.appendChild(qr);

React Example

import '@squidjam/qrcode-element';
import { useEffect, useRef, useState } from 'react';

function QRCodeDisplay() {
  const [text, setText] = useState('https://example.com');
  const qrRef = useRef(null);

  useEffect(() => {
    if (qrRef.current) {
      qrRef.current.addEventListener('qrcode-ready', () => {
        console.log('QR code rendered');
      });
    }
  }, []);

  return (
    <>>
      <input 
        value={text} 
        onChange={(e) => setText(e.target.value)} 
      />
      <qr-code
        ref={qrRef}
        data-text={text}
        data-width="256"
        data-height="256"
      />
    </>
  );
}

Vue 3 Example

<template>
  <div>
    <input v-model="url" />
    <qr-code
      :data-text="url"
      :data-width="size"
      :data-height="size"
      @qrcode-ready="onReady"
    />
  </div>
</template>

<script setup>
import '@squidjam/qrcode-element';
import { ref } from 'vue';

const url = ref('https://example.com');
const size = ref(256);

function onReady(event) {
  console.log('QR code ready:', event.detail);
}
</script>

Svelte Example

Angular Example

import { Component } from '@angular/core';
import '@squidjam/qrcode-element';

@Component({
  selector: 'app-qrcode',
  template: `
    <input [(ngModel)]="text" />
    <qr-code
      [attr.data-text]="text"
      [attr.data-width]="width"
      [attr.data-height]="height"
      (qrcode-ready)="onReady($event)"
    ></qr-code>
  `
})
export class QrcodeComponent {
  text = 'https://example.com';
  width = 256;
  height = 256;

  onReady(event: CustomEvent) {
    console.log('QR code ready:', event.detail);
  }
}

TypeScript Usage

import { QRCodeElement, QRCode, renderQRCode } from '@squidjam/qrcode-element';

// Use the element
const qr = document.createElement('qr-code') as QRCodeElement;
qr.text = 'Hello';
qr.width = 256;

// Or use the QRCode class directly
const qrcode = new QRCode(0, 0);
qrcode.addData('Hello World');
qrcode.make();

// Or use the render function
const element = renderQRCode({
  text: 'Hello',
  width: 256,
  height: 256,
  typeNumber: 0,
  errorCorrectLevel: 0,
  foreground: '#000000',
  background: '#ffffff',
  render: 'canvas'
});

import dynamic from 'next/dynamic';

const QRCode = dynamic(

  { ssr: false } // Prevent SSR since this is a browser API
);

export default function Page() {
  useEffect(() => {
    // This ensures the component only loads on client
  }, []);

  return (
    <>
      <script src="https://unpkg.com/@squidjam/qrcode-element" />
      <qr-code data-text="Next.js" />
    </>
  );
}
